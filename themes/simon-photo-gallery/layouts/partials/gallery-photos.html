{{- $photos := slice -}} {{- $favourites := slice -}} {{- $regular := slice -}} {{- range $index, $photo := hugo.Data.gallery.photos -}} {{- $source := $photo.src | default $photo.source | default $photo.urls.original | default $photo.urls.detail | default $photo.urls.grid -}} {{- if not $source }}{{ errorf "gallery photo at index %d is missing src" $index }}{{ end -}} {{- $filename := path.Base (urls.Parse $source).Path -}} {{- $stem := strings.TrimSuffix (path.Ext $filename) $filename -}} {{- $id := $photo.id | default ($stem | anchorize) -}} {{- $title := $photo.title | default $stem -}} {{- $alt := $photo.alt | default (printf "%s — photograph by %s" $title (site.Params.photographer | default site.Title)) -}} {{- $date := $photo.date | default "unknown" -}} {{- $location := $photo.location | default "unknown" -}} {{- $dimensions := $photo.dimensions | default (dict "width" 1500 "height" 1000) -}} {{- $urls := $photo.urls | default dict -}} {{- $urls = merge (dict "thumbnail" $source "grid" $source "detail" $source "original" $source) $urls -}} {{- $layout := $photo.layout | default dict -}} {{- $record := merge $photo (dict "id" $id "title" $title "alt" $alt "date" $date "location" $location "dimensions" $dimensions "urls" $urls "layout" $layout) -}} {{- if $photo.favourite -}} {{- $favourites = $favourites | append $record -}} {{- else -}} {{- $regular = $regular | append $record -}} {{- end -}} {{- end -}} {{- $photos = $favourites | append $regular -}} {{- return $photos -}}
